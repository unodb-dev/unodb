50// The byte arrays below are deliberately not `inline`, unlike every other
51// constant in this header. `test_values` constant-initializes a `std::span`
52// over each of them, and under `_GLIBCXX_DEBUG` plus UBSan, GCC 14.2's
53// constexpr evaluator cannot fold the sanitizer-instrumented `ptr + count`
54// inside `__glibcxx_requires_valid_range` when `ptr` is the address of an
55// inline (vague-linkage) variable - "is not a constant expression" - while it
56// folds the same expression over an internal-linkage array. Reproduced with
57// Ubuntu 24.04's g++-14 14.2.0-4ubuntu2~24.04.1; Homebrew GCC 14.4 accepts
58// both. micro_benchmark_utils.hpp's `values` relies on the same shape.

185[[nodiscard]] constexpr unodb::key_view copy_key(
186 unodb::key_view kv,
187 std::span<std::byte> buf UNODB_DETAIL_LIFETIMEBOUND) noexcept {
188 UNODB_DETAIL_ASSERT(kv.size() <= buf.size());
189 std::ranges::copy(kv, buf.begin());
190 return {buf.data(), kv.size()};

198[[nodiscard]] inline unodb::key_view make_key_17_byte10(
199 unodb::key_encoder& enc UNODB_DETAIL_LIFETIMEBOUND, std::uint8_t byte10,
200 std::uint8_t last) {
201 enc.reset();
202 for (unsigned i = 0; i < 10; ++i) enc.encode(std::uint8_t{0xAA});
203 enc.encode(byte10);
204 for (unsigned i = 11; i < 16; ++i) enc.encode(std::uint8_t{0xAA});
205 enc.encode(last);
206 return enc.get_key_view();
207}
