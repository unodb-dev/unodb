366inline void ensure_capacity(std::byte*& buf, size_t& cap, size_t off,
367 size_t min_capacity) {
368 // Find the smallest power of two >= min_capacity.
369 const auto asize = std::bit_ceil(min_capacity);
370 auto* tmp = detail::allocate_aligned(asize); // new allocation.
371 std::memcpy(tmp, buf, off); // copy over the data.
372 if (cap > INITIAL_BUFFER_CAPACITY) { // free old buffer iff allocated
373 detail::free_aligned(buf);
374 }
375 buf = static_cast<std::byte*>(tmp);
376 cap = asize;
377}

672 key_encoder& encode_text(std::span<const std::byte> text) {
673 // truncate view to at most maxlen bytes.
674 text = (text.size_bytes() > maxlen) ? text.subspan(0, maxlen) : text;
675 // normalize padding by stripping off any trailing [pad] bytes.
676 auto sz = text.size_bytes();
677 for (; sz > 0; sz--) {
678 if (text[sz - 1] != pad) break;
679 }
680 text = text.subspan(0, sz); // adjust span in case truncated.
681 // Ensure enough room for the text, the pad byte, and the
682 // run-length encoding of the pad byte.
683 ensure_available(sz + 1 + sizeof(size_type));
684 const auto padlen{static_cast<size_type>(maxlen - sz)};
685 append_bytes(text); // append bytes to the buffer.
686 encode(static_cast<std::uint8_t>(pad)); // encode as unsigned byte
687 encode(padlen); // logical run-length of the pad byte.
688 return *this;
689 }
