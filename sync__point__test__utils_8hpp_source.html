82template <unsigned ParkHit = 1, class Parked, class Interleaved>
83[[nodiscard]] unsigned run_parked_interleaving(unodb::detail::sync_point& point,
84 const Parked& parked,
85 const Interleaved& interleaved) {
86 static_assert(ParkHit > 0, "hits are 1-based; 0 would never park");
87 // Declared before the guard, so that the guard disarms the hook before the
88 // counter and the handshake it captures die. The handshake is local, so no
89 // earlier call or callable can leave it signalled, and ~thread_sync checks
90 // on return that this call left it reset.
91 unsigned hits = 0;
92 unodb::detail::thread_sync park_sync;
93 unodb::detail::thread_sync resume_sync;
94 const sync_point_guard guard{point};
95 point.arm([&hits, &park_sync, &resume_sync] {
96 if (++hits != ParkHit) return;
97 park_sync.notify();
98 resume_sync.wait();
99 });
100
101 unodb::this_thread().qsbr_pause();
102 // Spawned first, so that it is registered with QSBR before parked's thread
103 // exists: otherwise parked could start as the only registered thread, and
104 // its deallocations before the park would be immediate instead of deferred.
105 auto interleaved_thread =
106 unodb::qsbr_thread([&interleaved, &park_sync, &resume_sync] {
107 const unodb::quiescent_state_on_scope_exit q{};
108 park_sync.wait();
109 interleaved();
110 resume_sync.notify();
111 });
112 // noexcept when its body is, as GCC's -Wnoexcept wants: parked may be, and
113 // without stats so is ~quiescent_state_on_scope_exit.
114 auto parked_thread = unodb::qsbr_thread(
115 [&parked]() noexcept(noexcept(
116 unodb::quiescent_state_on_scope_exit{}) && noexcept(parked())) {
117 const unodb::quiescent_state_on_scope_exit q{};
118 parked();
119 });
120 parked_thread.join();
121 // interleaved has not run yet if parked never parked, and otherwise made all
122 // its hits before releasing parked, so this read races with nothing.
123 const bool missed_park = hits < ParkHit;
124 // Some interleaved operations pass point too.
125 if (missed_park) release_after_missed_park(park_sync, resume_sync, point);
126 interleaved_thread.join();
127 unodb::this_thread().qsbr_resume();
128 unodb::this_thread().quiescent();
129 if (missed_park) {
130 ADD_FAILURE() << "parked returned before hit " << ParkHit
131 << " of its sync point";
132 }
133 return hits;
134}
