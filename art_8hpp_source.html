607 [[nodiscard]] int cmp(art_key_type akey) const noexcept {
608 UNODB_DETAIL_ASSERT(!stack_.empty());
609 if constexpr (art_policy::full_key_in_inode_path) {
610 return unodb::detail::compare(keybuf_.get_key_view(),
611 akey.get_key_view());

614 UNODB_DETAIL_ASSERT(node.type() == node_type::LEAF);
615 const auto* const leaf{node.template ptr<leaf_type*>()};
616 return unodb::detail::compare(leaf->get_key_view(),

636 void push(detail::node_ptr node, std::byte key_byte,
637 std::uint8_t child_index, detail::key_prefix_snapshot prefix) {
638 // For variable length keys we need to know the number of bytes associated
639 // with the node's key_prefix. In addition there is one byte for the
640 // descent to the child node along the child_index. That information needs
641 // to be stored on the stack so we can pop off the right number of bytes
642 // even for OLC where the node might be concurrently modified.
643 UNODB_DETAIL_ASSERT(node.type() != node_type::LEAF);
644 stack_.push({node, key_byte, child_index, prefix});
645 keybuf_.push(prefix.get_key_view());
646 keybuf_.push(key_byte);
647 }
648
652 void push_leaf(detail::node_ptr aleaf) {
653 stack_.push({
654 aleaf,
655 static_cast<std::byte>(0xFFU), // ignored for leaf
656 static_cast<std::uint8_t>(0xFFU), // ignored for leaf
657 detail::key_prefix_snapshot(0), // ignored for leaf
658 true // packed_leaf
659 });
660 // No change in the key_buffer.
661 }

671 void push(const typename inode_base::iter_result& e) {
672 UNODB_DETAIL_ASSERT(e.node != nullptr);
673 const auto node_type = e.node.type();
674 if (UNODB_DETAIL_UNLIKELY(node_type == node_type::LEAF)) {
675 push_leaf(e.node);
676 return;
677 }
678 push(e.node, e.key_byte, e.child_index, e.prefix);
679 }

682 void pop() noexcept {
683 UNODB_DETAIL_ASSERT(!empty());
684
685 const auto& e = top();
686 const auto n = static_cast<std::size_t>(
687 (!(art_policy::can_eliminate_leaf && e.packed_leaf) &&
688 e.node.type() != node_type::LEAF)
689 ? e.prefix.length() + 1
690 : 0);
691 keybuf_.pop(n);
692 stack_.pop();
693 }

1380template <typename Key, typename Value, class INode>
1381detail::node_ptr* impl_helpers::add_or_choose_subtree(
1382 INode& inode, std::byte key_byte, basic_art_key<Key> k, Value v,
1383 db<Key, Value>& db_instance, tree_depth<basic_art_key<Key>> depth,
1384 detail::node_ptr* node_in_parent) {
1385 auto* const child =
1386 unwrap_fake_critical_section(inode.find_child(key_byte).second);
1387
1388 if (child != nullptr) return child;
1389
1390 if constexpr (art_policy<Key, Value>::can_eliminate_leaf) {
1391 // Value-in-slot: pack value directly, no leaf allocation.
1392 const auto packed = art_policy<Key, Value>::pack_value(v);
1393 const auto children_count = inode.get_children_count();
1394
1395 if constexpr (!std::is_same_v<INode, inode_256<Key, Value>>) {
1396 if (UNODB_DETAIL_UNLIKELY(children_count == INode::capacity)) {
1397 if constexpr (art_policy<Key, Value>::full_key_in_inode_path) {
1398 const auto chain_start =
1399 static_cast<tree_depth<basic_art_key<Key>>>(depth + 1);
1400 if (chain_start < k.size()) {
1401 // OOM safety: build chain BEFORE creating the larger node.
1402 auto chain_top = db_instance.build_chain(k, packed, chain_start);
1403 typename art_policy<Key, Value>::subtree_guard chain_guard{
1404 chain_top, db_instance};
1405 auto larger_node{INode::larger_derived_type::create(
1406 db_instance, inode, chain_top, depth, key_byte)};
1407 *node_in_parent = node_ptr{larger_node.release(),
1408 INode::larger_derived_type::type};
1409 chain_guard.release();
1410#ifdef UNODB_DETAIL_WITH_STATS
1411 db_instance.template account_growing_inode<
1412 INode::larger_derived_type::type>();
1413#endif
1414 // chain_top passed as node_ptr → value bit was set; clear it.
1415 auto& new_inode =
1416 *node_in_parent
1417 ->template ptr<typename INode::larger_derived_type*>();
1418 auto [ci, slotraw] = new_inode.find_child(key_byte);
1419 new_inode.clear_value_bit(ci);
1420 return child;
1421 }
1422 }
1423 auto larger_node{INode::larger_derived_type::create(
1424 db_instance, inode, packed, depth, key_byte)};
1425 *node_in_parent =
1426 node_ptr{larger_node.release(), INode::larger_derived_type::type};
1427#ifdef UNODB_DETAIL_WITH_STATS
1428 db_instance
1429 .template account_growing_inode<INode::larger_derived_type::type>();
1430#endif
1431 return child;
1432 }
1433 }
1434 if constexpr (art_policy<Key, Value>::full_key_in_inode_path) {
1435 const auto chain_start =
1436 static_cast<tree_depth<basic_art_key<Key>>>(depth + 1);
1437 if (chain_start < k.size()) {
1438 // OOM safety: build chain BEFORE mutating the tree. If
1439 // build_chain throws, the tree is unchanged.
1440 const auto chain_top = db_instance.build_chain(k, packed, chain_start);
1441 // Insert the packed value first (sets value bit correctly),
1442 // then overwrite the slot with the chain top and clear the bit.
1443 inode.add_to_nonfull(packed, depth, key_byte, children_count);
1444 std::atomic_signal_fence(std::memory_order_acq_rel);
1445 auto [ci2, slotraw2] = inode.find_child(key_byte);
1446 auto* const slot = unwrap_fake_critical_section(slotraw2);
1447 UNODB_DETAIL_ASSERT(slot != nullptr);
1448 *slot = chain_top;
1449 inode.clear_value_bit(ci2);
1450 return child;
1451 }
1452 }
1453 inode.add_to_nonfull(packed, depth, key_byte, children_count);
1454 return child;
1455 } else {
1456 // Leaf-based: allocate leaf as before.
1457 auto leaf = art_policy<Key, Value>::make_db_leaf_ptr(k, v, db_instance);
1458 const auto children_count = inode.get_children_count();
1459
1460 if constexpr (!std::is_same_v<INode, inode_256<Key, Value>>) {
1461 if (UNODB_DETAIL_UNLIKELY(children_count == INode::capacity)) {
1462 if constexpr (art_policy<Key, Value>::full_key_in_inode_path) {
1463 const auto chain_start =
1464 static_cast<tree_depth<basic_art_key<Key>>>(depth + 1);
1465 if (chain_start < k.size()) {
1466 // OOM safety: build chain BEFORE creating the larger node.
1467 const auto leaf_ptr =
1468 detail::node_ptr{leaf.release(), node_type::LEAF};
1469 auto chain_top = db_instance.build_chain(k, leaf_ptr, chain_start);
1470 typename art_policy<Key, Value>::subtree_guard chain_guard{
1471 chain_top, db_instance};
1472 auto larger_node{INode::larger_derived_type::create(
1473 db_instance, inode, chain_top, depth, key_byte)};
1474 *node_in_parent = node_ptr{larger_node.release(),
1475 INode::larger_derived_type::type};
1476 chain_guard.release();
1477#ifdef UNODB_DETAIL_WITH_STATS
1478 db_instance.template account_growing_inode<
1479 INode::larger_derived_type::type>();
1480#endif // UNODB_DETAIL_WITH_STATS
1481 return child;
1482 }
1483 }
1484 auto larger_node{INode::larger_derived_type::create(
1485 db_instance, inode, std::move(leaf), depth, key_byte)};
1486 *node_in_parent =
1487 node_ptr{larger_node.release(), INode::larger_derived_type::type};
1488#ifdef UNODB_DETAIL_WITH_STATS
1489 db_instance
1490 .template account_growing_inode<INode::larger_derived_type::type>();
1491#endif // UNODB_DETAIL_WITH_STATS
1492 return child;
1493 }
1494 }
1495 if constexpr (art_policy<Key, Value>::full_key_in_inode_path) {
1496 const auto chain_start =
1497 static_cast<tree_depth<basic_art_key<Key>>>(depth + 1);
1498 if (chain_start < k.size()) {
1499 // OOM safety: build chain BEFORE mutating the tree.
1500 const auto leaf_ptr = detail::node_ptr{leaf.release(), node_type::LEAF};
1501 const auto chain_top =
1502 db_instance.build_chain(k, leaf_ptr, chain_start);
1503 // node_ptr overload of add_to_nonfull; value bitmask is disabled
1504 // when can_eliminate_leaf is false, so set_value_bit is a no-op.
1505 inode.add_to_nonfull(chain_top, depth, key_byte, children_count);
1506 return child;
1507 }
1508 }
1509 inode.add_to_nonfull(std::move(leaf), depth, key_byte, children_count);
1510
1511 return child;
1512 } // else (leaf-based)
1513}

1516// on db param, but release() transfers ownership and the raw pointer's validity
1517// is independent of the temporary unique_ptr.
1518UNODB_DETAIL_DISABLE_MSVC_WARNING(26815)
1519template <typename Key, typename Value, class INode>
1520std::optional<detail::node_ptr*> impl_helpers::remove_or_choose_subtree(
1521 INode& inode, std::byte key_byte, basic_art_key<Key> k,
1522 db<Key, Value>& db_instance, detail::node_ptr* node_in_parent) {
1523 const auto [child_i, child_ptr]{inode.find_child(key_byte)};
1524
1525 if (child_ptr == nullptr) return {};
1526
1527 const auto child_ptr_val{child_ptr->load()};
1528
1529 if constexpr (art_policy<Key, Value>::can_eliminate_leaf) {
1530 if (!inode.is_value_in_slot(child_i)) {
1531 return unwrap_fake_critical_section(child_ptr);
1532 }
1533 } else {
1534 if (child_ptr_val.type() != node_type::LEAF)
1535 return unwrap_fake_critical_section(child_ptr);
1536
1537 const auto* const leaf{
1538 child_ptr_val.template ptr<typename db<Key, Value>::leaf_type*>()};
1539 if (!leaf->matches(k)) return {};
1540 }
1541
1542 if (UNODB_DETAIL_UNLIKELY(inode.is_min_size())) {
1543 if constexpr (std::is_same_v<INode, inode_4<Key, Value>>) {
1544 if (UNODB_DETAIL_LIKELY(inode.can_collapse(child_i))) {
1545 auto current_node{art_policy<Key, Value>::make_db_inode_unique_ptr(
1546 &inode, db_instance)};
1547 *node_in_parent = current_node->leave_last_child(child_i, db_instance);
1548#ifdef UNODB_DETAIL_WITH_STATS
1549 db_instance.template account_shrinking_inode<INode::type>();
1550#endif
1551 } else {
1552 // Prefix overflow — cannot collapse. Just remove the child entry.
1553 inode.remove(child_i, db_instance);
1554 }
1555 } else {
1556 auto new_node{
1557 INode::smaller_derived_type::create(db_instance, inode, child_i)};
1558 *node_in_parent =
1559 node_ptr{new_node.release(), INode::smaller_derived_type::type};
1560#ifdef UNODB_DETAIL_WITH_STATS
1561 db_instance.template account_shrinking_inode<INode::type>();
1562#endif
1563 }
1564 return nullptr;
1565 }
1566
1567 inode.remove(child_i, db_instance);
1568 return nullptr;
1569}

1580typename db<Key, Value>::get_result db<Key, Value>::get_internal(
1581 art_key_type k) const noexcept {
1582 if (UNODB_DETAIL_UNLIKELY(root == nullptr)) return {};
1583 if constexpr (std::is_same_v<Key, key_view>) {
1584 if (UNODB_DETAIL_UNLIKELY(k.size() == 0)) return {};
1585 }
1586
1587 auto node{root};
1588 auto remaining_key{k};
1589
1590 while (true) {
1591 const auto node_type = node.type();
1592 if (node_type == node_type::LEAF) {
1593 const auto* const leaf{node.template ptr<leaf_type*>()};
1594 if constexpr (art_policy::can_eliminate_key_in_leaf) {
1595 if (remaining_key.size() == 0)
1596 return leaf->template get_value<value_type>();
1597 } else {
1598 if (leaf->matches(k)) return leaf->template get_value<value_type>();
1599 }
1600 return {};
1601 }
1602
1603 UNODB_DETAIL_ASSERT(node_type != node_type::LEAF);
1604
1605 auto* const inode{node.template ptr<inode_type*>()};
1606 const auto& key_prefix{inode->get_key_prefix()};
1607 const auto key_prefix_length{key_prefix.length()};
1608 if (key_prefix.get_shared_length(remaining_key) < key_prefix_length)
1609 return {};
1610 remaining_key.shift_right(key_prefix_length);
1611 const auto [child_i,
1612 child_ptr]{inode->find_child(node_type, remaining_key[0])};
1613 if (child_ptr == nullptr) return {};
1614
1615 if constexpr (art_policy::can_eliminate_leaf) {
1616 if (inode->is_value_in_slot(node_type, child_i)) {
1617 if (remaining_key.size() != 1) return {};
1618 return art_policy::unpack_value(child_ptr->load());
1619 }
1620 }
1621
1622 node = *child_ptr;
1623 remaining_key.shift_right(1);
1624 }
1625}

1761template <typename Key, typename Value>
1762bool db<Key, Value>::insert_internal_key_view(art_key_type insert_key,
1763 value_type v) {
1764 auto* node = &root;
1765 tree_depth_type depth{};
1766 auto remaining_key{insert_key};
1767
1768 while (true) {
1769 if constexpr (art_policy::can_eliminate_leaf) {
1770 // Value-in-slot: no LEAF nodes in the tree. The insert loop
1771 // only descends through inodes. Packed values are detected
1772 // by is_value_in_slot after add_or_choose_subtree above.
1773 } else {
1774 const auto node_type = node->type();
1775 if (node_type == node_type::LEAF) {
1776 if constexpr (art_policy::can_eliminate_key_in_leaf) {
1777 // Keyless leaf: the inode path consumed all bytes of the existing
1778 // key. The ART prefix restriction guarantees no key is a prefix
1779 // of another, so remaining_key must be empty → duplicate.
1780 UNODB_DETAIL_ASSERT(remaining_key.size() == 0);
1781 return false;
1782 } else { // LCOV_EXCL_START — dead for can_eliminate_key_in_leaf
1783 auto* const leaf{node->template ptr<leaf_type*>()};
1784 const auto existing_key{leaf->get_key_view()};
1785 const auto cmp = insert_key.cmp(existing_key);
1786 if (UNODB_DETAIL_UNLIKELY(cmp == 0)) {
1787 return false; // exists
1788 }
1789 constexpr auto cap = detail::key_prefix_capacity;
1790 const auto remaining_existing = existing_key.subspan(depth);
1791 const auto shared = detail::key_prefix_snapshot::shared_len(
1792 detail::get_u64(remaining_existing), remaining_key.get_u64(),
1793 cap);
1794 if (shared >= cap && remaining_existing.size() > cap &&
1795 remaining_key.size() > cap &&
1796 remaining_existing[cap] == remaining_key[cap]) {
1797 const auto dispatch = remaining_existing[cap];
1798 auto chain{inode_4::create(*this, existing_key, remaining_key,
1799 depth, dispatch, *node)};
1800 *node = detail::node_ptr{chain.release(), node_type::I4};

2299typename db<Key, Value>::iterator& db<Key, Value>::iterator::seek(
2300 art_key_type search_key, bool& match, bool fwd) {
2301 invalidate(); // invalidate the iterator (clear the stack).
2302 match = false; // unless we wind up with an exact match.
2303 if (UNODB_DETAIL_UNLIKELY(db_.root == nullptr)) return *this; // aka end
2304
2305 // Empty key_view sorts before all keys — go to first (fwd) or end (rev).
2306 if constexpr (std::is_same_v<Key, key_view>) {
2307 if (UNODB_DETAIL_UNLIKELY(search_key.size() == 0)) {
2308 return fwd ? left_most_traversal(db_.root) : *this;
2309 }
2310 }
2311
2312 auto node{db_.root};
2313 const auto k = search_key;
2314 auto remaining_key{k};
2315
2316 while (true) {
2317 const auto node_type = node.type();
2318 if constexpr (!art_policy::can_eliminate_leaf) {
2319 if (node_type == node_type::LEAF) {
2320 const auto* const leaf{node.template ptr<leaf_type*>()};
2321 push_leaf(node);
2322 int cmp_{0};
2323 if constexpr (art_policy::full_key_in_inode_path) {
2324 cmp_ =
2325 unodb::detail::compare(keybuf_.get_key_view(), k.get_key_view());
2326 } else {
2327 cmp_ = leaf->cmp(k);
2328 }
2329 if (cmp_ == 0) {

2331 return *this;
2332 }
2333 if (fwd) { // GTE semantics
2334 // if search_key < leaf, use the leaf, else next().
2335 return (cmp_ < 0) ? *this : next();
2336 }
2337 // LTE semantics: if search_key > leaf, use the leaf, else prior().
2338 return (cmp_ > 0) ? *this : prior();
2339 }
2340 } // if constexpr (!can_eliminate_leaf)
2341 UNODB_DETAIL_ASSERT(node_type != node_type::LEAF);
2342 auto* const inode{node.template ptr<inode_type*>()}; // some internal node.
2343 const auto key_prefix{inode->get_key_prefix().get_snapshot()}; // prefix
2344 const auto key_prefix_length{key_prefix.length()}; // length of that prefix
2345 const auto shared_length = key_prefix.get_shared_length(
2346 remaining_key.get_u64()); // #of prefix bytes matched.
2347 if (shared_length < key_prefix_length) {
2348 // We have visited an internal node whose prefix is longer than
2349 // the bytes in the key that we need to match. To figure out
2350 // whether the search key would be located before or after the
2351 // current internal node, we need to compare the respective key
2352 // spans lexicographically. Since we have [shared_length] bytes
2353 // in common, we know that the next byte will tell us the
2354 // relative ordering of the key vs the prefix. So now we compare
2355 // prefix and key and the first byte where they differ.
2356 const auto cmp_ = static_cast<int>(remaining_key[shared_length]) -
2357 static_cast<int>(key_prefix[shared_length]);
2358 UNODB_DETAIL_ASSERT(cmp_ != 0);

2360 if (cmp_ < 0) {
2361 // FWD and the search key is ordered before this node. We
2362 // want the left-most leaf under the node.
2363 return left_most_traversal(node);
2364 }
2365 // FWD and the search key is ordered after this node. Right
2366 // most descent and then next().
2367 return right_most_traversal(node).next();
2368 }
2369 // reverse traversal
2370 if (cmp_ < 0) {
2371 // REV and the search key is ordered before this node. We
2372 // want the preceding key.
2373 return left_most_traversal(node).prior();
2374 }
2375 // REV and the search key is ordered after this node.
2376 return right_most_traversal(node);
2377 }
2378 remaining_key.shift_right(key_prefix_length);
2379 // Key fully consumed by prefix — search key is a proper prefix of
2380 // all keys under this inode. Leftmost leaf is GTE, prior is LTE.
2381 if constexpr (std::is_same_v<Key, key_view>) {
2382 if (UNODB_DETAIL_UNLIKELY(remaining_key.size() == 0)) {
2383 return fwd ? left_most_traversal(node)
2384 : left_most_traversal(node).prior();
2385 }
2386 }
2387 const auto res = inode->find_child(node_type, remaining_key[0]);
2388 if (res.second == nullptr) {
2389 // We are on a key byte during the descent that is not mapped by
2390 // the current node. Where we go next depends on whether we are
2391 // doing forward or reverse traversal.
2392 if (fwd) {
2393 // FWD: Take the next child_index that is mapped in the data
2394 // and then do a left-most descent to land on the key that is
2395 // the immediate successor of the desired key in the data.
2396 //
2397 // Note: We are probing with a key byte which does not appear
2398 // in our list of keys (this was verified above) so this will
2399 // always be the index the first entry whose key byte is
2400 // greater-than the probe value and [false] if there is no
2401 // such entry.
2402 //
2403 // Note: [node] has not been pushed onto the stack yet!
2404 auto nxt = inode->gte_key_byte(node_type, remaining_key[0]);
2405 if (!nxt) {
2406 // Pop entries off the stack until we find one with a
2407 // right-sibling of the path we took to this node and then
2408 // do a left-most descent under that right-sibling. If there
2409 // is no such parent, we will wind up with an empty stack
2410 // (iterator at the end) and return that state.
2411 //
2412 // Note: The current node (where gte_key_byte failed) was
2413 // never pushed, so the stack top is already the first
2414 // ancestor to check for a right-sibling.
2415 while (!empty()) {
2416 const auto& centry = top();
2417 const auto cnode{centry.node}; // possible parent from the stack
2418 auto* const icnode{cnode.template ptr<inode_type*>()};
2419 const auto cnxt = icnode->next(
2420 cnode.type(), centry.child_index); // right-sibling.
2421 if (cnxt) {
2422 const auto& e2 = cnxt.value();
2423 pop();
2424 push(e2);
2425 auto nchild = icnode->get_child(cnode.type(), e2.child_index);
2426 return descend_left(icnode, cnode.type(), e2.child_index, nchild);
2427 }
2428 pop();
2429 }
2430 return *this; // stack is empty (aka end iterator).
2431 }
2432 const auto& tmp = nxt.value(); // unwrap.
2433 const auto child_index = tmp.child_index;
2434 const auto child = inode->get_child(node_type, child_index);
2435 push(node, tmp.key_byte, child_index, tmp.prefix); // the path we took
2436 return descend_left(inode, node_type, child_index, child);
2437 }
2438 // REV: Take the prior child_index that is mapped and then do
2439 // a right-most descent to land on the key that is the
2440 // immediate predecessor of the desired key in the data.
2441 auto nxt = inode->lte_key_byte(node_type, remaining_key[0]);
2442 if (!nxt) {
2443 // Pop off entries until we find one with a left-sibling and
2444 // then do a right-most descent under that left-sibling. In
2445 // the extreme case there is no such previous entry and we
2446 // will wind up with an empty stack.
2447 //
2448 // Note: The current node (where lte_key_byte failed) was
2449 // never pushed, so the stack top is already the first
2450 // ancestor to check for a left-sibling.
2451 while (!empty()) {
2452 const auto& centry = top();
2453 const auto cnode{centry.node}; // possible parent from stack
2454 auto* const icnode{cnode.template ptr<inode_type*>()};
2455 const auto cnxt =
2456 icnode->prior(cnode.type(), centry.child_index); // left-sibling.
2457 if (cnxt) {
2458 const auto& e2 = cnxt.value();
2459 pop();
2460 push(e2);
2461 auto nchild = icnode->get_child(cnode.type(), e2.child_index);
2462 return descend_right(icnode, cnode.type(), e2.child_index, nchild);
2463 }
2464 pop();
2465 }
2466 return *this; // stack is empty (aka end iterator).
2467 }
2468 const auto& tmp = nxt.value(); // unwrap.
2469 const auto child_index{tmp.child_index};
2470 const auto child = inode->get_child(node_type, child_index);
2471 push(node, tmp.key_byte, child_index, tmp.prefix); // the path we took
2472 return descend_right(inode, node_type, child_index, child);
2473 }
2474 // Simple case. There is a child for the current key byte.
2475 const auto child_index{res.first};
2476 const auto* const child{res.second};
2477 push(node, remaining_key[0], child_index, key_prefix);
2478 if constexpr (art_policy::can_eliminate_leaf) {
2479 if (inode->is_value_in_slot(node_type, child_index)) {
2480 push_leaf(*child);
2481 if (remaining_key.size() <= 1) {
2482 match = true;
2483 return *this;
2484 }
2485 // VIS key is a strict prefix of search key (VIS < search).
2486 return fwd ? next() : *this;
2487 }
2488 }
2489 node = *child;
2490 remaining_key.shift_right(1);
2491 } // while ( true )
2492 UNODB_DETAIL_CANNOT_HAPPEN();
2493}

2722template <typename Key, typename Value>
2723template <typename FN>
2724bool db<Key, Value>::upsert(Key k, value_type v, FN fn) {
2725 static_assert(std::is_invocable_r_v<upsert_action, FN, value_type&>,
2726 "upsert lambda must be callable as upsert_action(value_type&)");
2727
2728 const art_key_type art_k{k};
2729
2730 // Attempt insert. If key is absent this succeeds and we are done.
2731 if (insert_internal(art_k, v)) return true;
2732
2733 // Key exists. Traverse to find the value location.
2734 auto node{root};
2735 auto remaining_key{art_k};
2736
2737 while (true) {
2738 const auto node_type = node.type();
2739
2740 if constexpr (!art_policy::can_eliminate_leaf) {
2741 if (node_type == node_type::LEAF) {
2742 UNODB_DETAIL_DISABLE_MSVC_WARNING(26462)
2743 auto* const leaf{node.template ptr<leaf_type*>()};
2744 UNODB_DETAIL_RESTORE_MSVC_WARNINGS() // Read value into local copy,
2745 // call lambda.
2746 auto local = leaf->template get_value<value_type>();
2747 const auto action = fn(local);
2748 UNODB_DETAIL_DISABLE_MSVC_WARNING(6326)
2749 switch (action) {
2750 case upsert_action::keep:
2751 return false;
2752 case upsert_action::update:
2753 if constexpr (std::is_same_v<value_type, value_view>) {
2754 // Replace the leaf with a new one containing the proposed value.
2755 [[maybe_unused]] const auto removed = remove_internal(art_k);
2756 UNODB_DETAIL_ASSERT(removed);
2757 [[maybe_unused]] const auto inserted = insert_internal(art_k, v);
2758 UNODB_DETAIL_ASSERT(inserted);
2759 } else {
2760 leaf->template set_value<value_type>(local);
2761 }
2762 return false;
2763 case upsert_action::erase: {
2764 [[maybe_unused]] const auto removed = remove_internal(art_k);
2765 UNODB_DETAIL_ASSERT(removed);
2766 return false;
2767 }
2768 }
2769 UNODB_DETAIL_RESTORE_MSVC_WARNINGS()
2770 UNODB_DETAIL_CANNOT_HAPPEN(); // LCOV_EXCL_LINE
2771 }
2772 }
2773
2774 UNODB_DETAIL_ASSERT(node_type != node_type::LEAF);
2775
2776 auto* const inode{node.template ptr<inode_type*>()};
2777 const auto& key_prefix{inode->get_key_prefix()};
2778 const auto key_prefix_length{key_prefix.length()};
2779 UNODB_DETAIL_ASSERT(key_prefix.get_shared_length(remaining_key) ==
2780 key_prefix_length);
2781 remaining_key.shift_right(key_prefix_length);
2782
2783 const auto [child_i,
2784 child_ptr]{inode->find_child(node_type, remaining_key[0])};
2785 UNODB_DETAIL_ASSERT(child_ptr != nullptr);
2786
2787 if constexpr (art_policy::can_eliminate_leaf) {
2788 if (inode->is_value_in_slot(node_type, child_i)) {
2789 // VIS: unpack, call lambda, pack back.
2790 auto local = art_policy::unpack_value(child_ptr->load());
2791 const auto action = fn(local);
2792 UNODB_DETAIL_DISABLE_MSVC_WARNING(6326)
2793 switch (action) {
2794 case upsert_action::keep:
2795 return false;
2796 case upsert_action::update:
2797 if constexpr (std::is_same_v<value_type, value_view>) {
2798 UNODB_DETAIL_CANNOT_HAPPEN(); // LCOV_EXCL_LINE
2799 } else {
2800 *child_ptr = art_policy::pack_value(local);
2801 }
2802 return false;
2803 case upsert_action::erase: {
2804 [[maybe_unused]] const auto removed = remove_internal(art_k);
2805 UNODB_DETAIL_ASSERT(removed);
2806 return false;
2807 }
2808 }
2809 UNODB_DETAIL_RESTORE_MSVC_WARNINGS()
2810 UNODB_DETAIL_CANNOT_HAPPEN(); // LCOV_EXCL_LINE
2811 }
2812 }
2813
2814 node = child_ptr->load();
2815 remaining_key.shift_right(1);
2816 }
