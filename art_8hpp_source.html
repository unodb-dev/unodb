633 UNODB_DETAIL_DISABLE_MSVC_WARNING(26440)
634 [[nodiscard]] int cmp(art_key_type akey) const
635 noexcept(!art_policy::has_heap) {
636 UNODB_DETAIL_ASSERT(!stack_.empty());
637 if constexpr (art_policy::full_key_in_inode_path) {
638 return unodb::detail::compare(keybuf_.get_key_view(),
639 akey.get_key_view());
640 } else if constexpr (art_policy::has_heap) {
641 // Heap mode: recover full key into get_key_buf_ so get_key() can
642 // reuse it without a redundant extract_key call.
643 const auto& node = stack_.top().node;
644 const auto value_id = art_policy::unpack_value(node);
645 const auto kv = db_.heap_.heap.extract_key(value_id, get_key_buf_);
646 return unodb::detail::compare(kv, akey.get_key_view());
647 } else {
648 const auto& node = stack_.top().node;
649 UNODB_DETAIL_ASSERT(node.type() == node_type::LEAF);
650 const auto* const leaf{node.template ptr<leaf_type*>()};
651 return unodb::detail::compare(leaf->get_key_view(),
652 akey.get_key_view());
653 }
654 }

672 void push(detail::node_ptr node, std::byte key_byte,
673 std::uint8_t child_index, detail::key_prefix_snapshot prefix) {
674 // For variable length keys we need to know the number of bytes associated
675 // with the node's key_prefix. In addition there is one byte for the
676 // descent to the child node along the child_index. That information needs
677 // to be stored on the stack so we can pop off the right number of bytes
678 // even for OLC where the node might be concurrently modified.
679 UNODB_DETAIL_ASSERT(node.type() != node_type::LEAF);
680 stack_.push({node, key_byte, child_index, prefix});
681 keybuf_.push(prefix.get_key_view());
682 keybuf_.push(key_byte);
683 }
684
688 void push_leaf(detail::node_ptr aleaf) {
689 stack_.push({
690 aleaf,
691 static_cast<std::byte>(0xFFU), // ignored for leaf
692 static_cast<std::uint8_t>(0xFFU), // ignored for leaf
693 detail::key_prefix_snapshot(0), // ignored for leaf
694 art_policy::can_eliminate_leaf // is_packed_value
695 });
696 // No change in the key_buffer.
697 }
698
707 void push(const typename inode_base::iter_result& e) {
708 UNODB_DETAIL_ASSERT(e.node != nullptr);
709 push(e.node, e.key_byte, e.child_index, e.prefix);
710 }

713 void pop() noexcept {
714 UNODB_DETAIL_ASSERT(!empty());
715
716 const auto& e = top();
717 const auto n = static_cast<std::size_t>(
718 (!e.is_packed_value && e.node.type() != node_type::LEAF)
719 ? e.prefix.length() + 1
720 : 0);
721 keybuf_.pop(n);
722 stack_.pop();
723 }

1446template <typename Key, typename Value, typename HeapTag, class INode>
1447detail::node_ptr* impl_helpers::add_or_choose_subtree(
1448 INode& inode, std::byte key_byte, basic_art_key<Key> k, Value v,
1449 db<Key, Value, HeapTag>& db_instance, tree_depth<basic_art_key<Key>> depth,
1450 detail::node_ptr* node_in_parent) {
1451 auto* const child =
1452 unwrap_fake_critical_section(inode.find_child(key_byte).second);
1453
1454 if (child != nullptr) return child;
1455
1456 if constexpr (art_policy<Key, Value, HeapTag>::can_eliminate_leaf) {
1457 // Value-in-slot: pack value directly, no leaf allocation.
1458 const auto packed = art_policy<Key, Value, HeapTag>::pack_value(v);
1459 const auto children_count = inode.get_children_count();
1460
1461 if constexpr (!std::is_same_v<INode, inode_256<Key, Value, HeapTag>>) {
1462 if (UNODB_DETAIL_UNLIKELY(children_count == INode::capacity)) {
1463 if constexpr (art_policy<Key, Value, HeapTag>::full_key_in_inode_path) {
1464 const auto chain_start =
1465 static_cast<tree_depth<basic_art_key<Key>>>(depth + 1);
1466 if (chain_start < k.size()) {
1467 // OOM safety: build chain BEFORE creating the larger node.
1468 auto chain_top = db_instance.build_chain(k, packed, chain_start);
1469 typename art_policy<Key, Value, HeapTag>::subtree_guard chain_guard{
1470 chain_top, db_instance};
1471 auto larger_node{INode::larger_derived_type::create(
1472 db_instance, inode, chain_top, depth, key_byte)};
1473 *node_in_parent = node_ptr{larger_node.release(),
1474 INode::larger_derived_type::type};
1475 chain_guard.release();
1476#ifdef UNODB_DETAIL_WITH_STATS
1477 db_instance.template account_growing_inode<
1478 INode::larger_derived_type::type>();
1479#endif
1480 // chain_top passed as node_ptr → value bit was set; clear it.
1481 auto& new_inode =
1482 *node_in_parent
1483 ->template ptr<typename INode::larger_derived_type*>();
1484 auto [ci, slotraw] = new_inode.find_child(key_byte);
1485 new_inode.clear_value_bit(ci);
1486 return child;
1487 }
1488 }
1489 auto larger_node{INode::larger_derived_type::create(
1490 db_instance, inode, packed, depth, key_byte)};
1491 *node_in_parent =
1492 node_ptr{larger_node.release(), INode::larger_derived_type::type};
1493#ifdef UNODB_DETAIL_WITH_STATS
1494 db_instance
1495 .template account_growing_inode<INode::larger_derived_type::type>();
1496#endif
1497 return child;
1498 }
1499 }
1500 if constexpr (art_policy<Key, Value, HeapTag>::full_key_in_inode_path) {
1501 const auto chain_start =
1502 static_cast<tree_depth<basic_art_key<Key>>>(depth + 1);
1503 if (chain_start < k.size()) {
1504 // OOM safety: build chain BEFORE mutating the tree. If
1505 // build_chain throws, the tree is unchanged.
1506 const auto chain_top = db_instance.build_chain(k, packed, chain_start);
1507 // Insert the packed value first (sets value bit correctly),
1508 // then overwrite the slot with the chain top and clear the bit.
1509 inode.add_to_nonfull(packed, depth, key_byte, children_count);
1510 std::atomic_signal_fence(std::memory_order_acq_rel);
1511 auto [ci2, slotraw2] = inode.find_child(key_byte);
1512 auto* const slot = unwrap_fake_critical_section(slotraw2);
1513 UNODB_DETAIL_ASSERT(slot != nullptr);
1514 *slot = chain_top;
1515 inode.clear_value_bit(ci2);
1516 return child;
1517 }
1518 }
1519 inode.add_to_nonfull(packed, depth, key_byte, children_count);
1520 return child;
1521 } else {
1522 // Leaf-based: allocate leaf as before.
1523 auto leaf =
1524 art_policy<Key, Value, HeapTag>::make_db_leaf_ptr(k, v, db_instance);
1525 const auto children_count = inode.get_children_count();
1526
1527 if constexpr (!std::is_same_v<INode, inode_256<Key, Value, HeapTag>>) {
1528 if (UNODB_DETAIL_UNLIKELY(children_count == INode::capacity)) {
1529 if constexpr (art_policy<Key, Value, HeapTag>::full_key_in_inode_path) {
1530 const auto chain_start =
1531 static_cast<tree_depth<basic_art_key<Key>>>(depth + 1);
1532 if (chain_start < k.size()) {
1533 // OOM safety: build chain BEFORE creating the larger node.
1534 const auto leaf_ptr =
1535 detail::node_ptr{leaf.release(), node_type::LEAF};
1536 auto chain_top = db_instance.build_chain(k, leaf_ptr, chain_start);
1537 typename art_policy<Key, Value, HeapTag>::subtree_guard chain_guard{
1538 chain_top, db_instance};
1539 auto larger_node{INode::larger_derived_type::create(
1540 db_instance, inode, chain_top, depth, key_byte)};
1541 *node_in_parent = node_ptr{larger_node.release(),
1542 INode::larger_derived_type::type};
1543 chain_guard.release();
1544#ifdef UNODB_DETAIL_WITH_STATS
1545 db_instance.template account_growing_inode<
1546 INode::larger_derived_type::type>();
1547#endif // UNODB_DETAIL_WITH_STATS
1548 return child;
1549 }
1550 }
1551 auto larger_node{INode::larger_derived_type::create(
1552 db_instance, inode, std::move(leaf), depth, key_byte)};
1553 *node_in_parent =
1554 node_ptr{larger_node.release(), INode::larger_derived_type::type};
1555#ifdef UNODB_DETAIL_WITH_STATS
1556 db_instance
1557 .template account_growing_inode<INode::larger_derived_type::type>();
1558#endif // UNODB_DETAIL_WITH_STATS
1559 return child;
1560 }
1561 }
1562 if constexpr (art_policy<Key, Value, HeapTag>::full_key_in_inode_path) {
1563 const auto chain_start =
1564 static_cast<tree_depth<basic_art_key<Key>>>(depth + 1);
1565 if (chain_start < k.size()) {
1566 // OOM safety: build chain BEFORE mutating the tree.
1567 const auto leaf_ptr = detail::node_ptr{leaf.release(), node_type::LEAF};
1568 const auto chain_top =
1569 db_instance.build_chain(k, leaf_ptr, chain_start);
1570 // node_ptr overload of add_to_nonfull; value bitmask is disabled
1571 // when can_eliminate_leaf is false, so set_value_bit is a no-op.
1572 inode.add_to_nonfull(chain_top, depth, key_byte, children_count);
1573 return child;
1574 }
1575 }
1576 inode.add_to_nonfull(std::move(leaf), depth, key_byte, children_count);
1577
1578 return child;
1579 } // else (leaf-based)
1580}

1587std::optional<detail::node_ptr*> impl_helpers::remove_or_choose_subtree(
1588 INode& inode, std::byte key_byte, basic_art_key<Key> k,
1589 db<Key, Value, HeapTag>& db_instance, detail::node_ptr* node_in_parent) {
1590 const auto [child_i, child_ptr]{inode.find_child(key_byte)};
1591
1592 if (child_ptr == nullptr) return {};
1593
1594 const auto child_ptr_val{child_ptr->load()};

1596 if constexpr (art_policy<Key, Value, HeapTag>::can_eliminate_leaf) {
1597 if (!inode.is_value_in_slot(child_i)) {
1598 return unwrap_fake_critical_section(child_ptr);
1599 }
1600 } else {
1601 if (child_ptr_val.type() != node_type::LEAF)
1602 return unwrap_fake_critical_section(child_ptr);

1604 const auto* const leaf{
1605 child_ptr_val
1606 .template ptr<typename db<Key, Value, HeapTag>::leaf_type*>()};
1607 if (!leaf->matches(k)) return {};
1608 }
1609
1610 if (UNODB_DETAIL_UNLIKELY(inode.is_min_size())) {
1611 if constexpr (std::is_same_v<INode, inode_4<Key, Value, HeapTag>>) {
1612 if (UNODB_DETAIL_LIKELY(inode.can_collapse(child_i))) {
1613 auto current_node{
1614 art_policy<Key, Value, HeapTag>::make_db_inode_unique_ptr(
1615 &inode, db_instance)};
1616 *node_in_parent = current_node->leave_last_child(child_i, db_instance);
1617#ifdef UNODB_DETAIL_WITH_STATS
1618 db_instance.template account_shrinking_inode<INode::type>();
1619#endif
1620 } else {
1621 // Prefix overflow — cannot collapse. Just remove the child entry.
1622 inode.remove(child_i, db_instance);
1623 }
1624 } else {
1625 auto new_node{
1626 INode::smaller_derived_type::create(db_instance, inode, child_i)};
1627 *node_in_parent =
1628 node_ptr{new_node.release(), INode::smaller_derived_type::type};
1629#ifdef UNODB_DETAIL_WITH_STATS
1630 db_instance.template account_shrinking_inode<INode::type>();
1631#endif
1632 }
1633 return nullptr;
1634 }
1635
1636 inode.remove(child_i, db_instance);
1637 return nullptr;
1638}

1653 if (UNODB_DETAIL_UNLIKELY(k.size() == 0)) return {};

1831bool db<Key, Value, HeapTag>::insert_internal_key_view(art_key_type insert_key,
1832 value_type v) {
1833 auto* node = &root;
1834 tree_depth_type depth{};
1835 auto remaining_key{insert_key};
1836
1837 while (true) {
1838 if constexpr (art_policy::can_eliminate_leaf) {
1839 // Value-in-slot: no LEAF nodes in the tree. The insert loop
1840 // only descends through inodes. Packed values are detected
1841 // by is_value_in_slot after add_or_choose_subtree above.
1842 } else {
1843 const auto node_type = node->type();
1844 if (node_type == node_type::LEAF) {
1845 if constexpr (art_policy::can_eliminate_key_in_leaf) {
1846 // Keyless leaf: the inode path consumed all bytes of the existing
1847 // key. The ART prefix restriction guarantees no key is a prefix
1848 // of another, so remaining_key must be empty → duplicate.
1849 UNODB_DETAIL_ASSERT(remaining_key.size() == 0);
1850 return false;
1851 } else { // LCOV_EXCL_START — dead for can_eliminate_key_in_leaf
1852 auto* const leaf{node->template ptr<leaf_type*>()};
1853 const auto existing_key{leaf->get_key_view()};
1854 const auto cmp = insert_key.cmp(existing_key);
1855 if (UNODB_DETAIL_UNLIKELY(cmp == 0)) {
1856 return false; // exists
1857 }
1858 constexpr auto cap = detail::key_prefix_capacity;
1859 const auto remaining_existing = existing_key.subspan(depth);
1860 const auto shared = detail::key_prefix_snapshot::shared_len(
1861 detail::get_u64(remaining_existing), remaining_key.get_u64(),
1862 cap);
1863 if (shared >= cap && remaining_existing.size() > cap &&
1864 remaining_key.size() > cap &&
1865 remaining_existing[cap] == remaining_key[cap]) {
1866 const auto dispatch = remaining_existing[cap];
1867 auto chain{inode_4::create(*this, existing_key, remaining_key,
1868 depth, dispatch, *node)};
1869 *node = detail::node_ptr{chain.release(), node_type::I4};
1870#ifdef UNODB_DETAIL_WITH_STATS
1871 account_growing_inode<node_type::I4>();
1872#endif // UNODB_DETAIL_WITH_STATS
1873 continue;
1874 }
1875 auto new_leaf = art_policy::make_db_leaf_ptr(insert_key, v, *this);
1876 // full_key_in_inode_path == can_eliminate_key_in_leaf, so
1877 // inside !can_eliminate_key_in_leaf the chain path is dead.
1878 static_assert(!art_policy::full_key_in_inode_path);
1879 auto new_node{inode_4::create(*this, existing_key, remaining_key,
1880 depth, leaf, std::move(new_leaf))};
1881 *node = detail::node_ptr{new_node.release(), node_type::I4};
1882#ifdef UNODB_DETAIL_WITH_STATS
1883 account_growing_inode<node_type::I4>();
1884#endif // UNODB_DETAIL_WITH_STATS
1885
1886 return true;
1887 } // else (keyed leaf) LCOV_EXCL_STOP
1888 }
1889 } // else (!can_eliminate_leaf)
1890
1891 const auto node_type = node->type();
1892 UNODB_DETAIL_ASSERT(node_type != node_type::LEAF);
1893
1894 auto* const inode{node->template ptr<inode_type*>()};
1895 const auto& key_prefix{inode->get_key_prefix()};
1896 const auto key_prefix_length{key_prefix.length()};
1897 const auto shared_prefix_len{key_prefix.get_shared_length(remaining_key)};
1898 if (shared_prefix_len < key_prefix_length) {
1899 if constexpr (art_policy::full_key_in_inode_path) {
1900 const auto chain_start =
1901 static_cast<tree_depth_type>(depth + shared_prefix_len + 1);
1902 if (chain_start < insert_key.size()) {
1903 // OOM safety: build chain BEFORE creating the prefix-split I4.
1904 if constexpr (art_policy::can_eliminate_leaf) {
1905 auto chain_top =
1906 build_chain(insert_key, art_policy::pack_value(v), chain_start);
1907 typename art_policy::subtree_guard chain_guard{chain_top, *this};
1908 auto new_node =
1909 inode_4::create(*this, *node, shared_prefix_len, depth,
1910 chain_top, remaining_key[shared_prefix_len]);
1911 *node = detail::node_ptr{new_node.release(), node_type::I4};
1912 chain_guard.release();
1913 // chain_top passed as node_ptr → value bit set; clear it.
1914 auto* const new_inode = node->template ptr<inode_type*>();
1915 auto [ci4, slotraw4] = new_inode->find_child(
1916 node_type::I4, remaining_key[shared_prefix_len]);
1917 new_inode->clear_value_bit(node_type::I4, ci4);
1918 } else {
1919 auto leaf = art_policy::make_db_leaf_ptr(insert_key, v, *this);
1920 const auto leaf_ptr =
1921 detail::node_ptr{leaf.release(), node_type::LEAF};
1922 auto chain_top = build_chain(insert_key, leaf_ptr, chain_start);
1923 typename art_policy::subtree_guard chain_guard{chain_top, *this};
1924 auto new_node =
1925 inode_4::create(*this, *node, shared_prefix_len, depth,
1926 chain_top, remaining_key[shared_prefix_len]);
1927 *node = detail::node_ptr{new_node.release(), node_type::I4};
1928 chain_guard.release();
1929 }
1930#ifdef UNODB_DETAIL_WITH_STATS
1931 account_growing_inode<node_type::I4>();
1932 ++key_prefix_splits;
1933 UNODB_DETAIL_ASSERT(
1934 growing_inode_counts[internal_as_i<node_type::I4>] >
1935 key_prefix_splits);
1936#endif // UNODB_DETAIL_WITH_STATS
1937 return true;
1938 }
1939 }
1940 // No chain needed (short key or !full_key_in_inode_path).
1941 if constexpr (art_policy::can_eliminate_leaf) {
1942 auto new_node = inode_4::create(*this, *node, shared_prefix_len, depth,
1943 art_policy::pack_value(v),
1944 remaining_key[shared_prefix_len]);
1945 *node = detail::node_ptr{new_node.release(), node_type::I4};
1946 } else {
1947 auto leaf = art_policy::make_db_leaf_ptr(insert_key, v, *this);
1948 auto new_node =
1949 inode_4::create(*this, *node, shared_prefix_len, depth,
1950 std::move(leaf), remaining_key[shared_prefix_len]);
1951 *node = detail::node_ptr{new_node.release(), node_type::I4};
1952 }
1953#ifdef UNODB_DETAIL_WITH_STATS
1954 account_growing_inode<node_type::I4>();
1955 ++key_prefix_splits;
1956 UNODB_DETAIL_ASSERT(growing_inode_counts[internal_as_i<node_type::I4>] >
1957 key_prefix_splits);
1958#endif // UNODB_DETAIL_WITH_STATS
1959 return true;
1960 }
1961 UNODB_DETAIL_ASSERT(shared_prefix_len == key_prefix_length);
1962 depth += key_prefix_length;
1963 remaining_key.shift_right(key_prefix_length);
1964
1965 node = inode->template add_or_choose_subtree<detail::node_ptr*>(
1966 node_type, remaining_key[0], insert_key, v, *this, depth, node);
1967
1968 if (node == nullptr) return true;
1969
1970 if constexpr (art_policy::can_eliminate_leaf) {
1971 const auto [ci, _] = inode->find_child(node_type, remaining_key[0]);
1972 if (inode->is_value_in_slot(node_type, ci)) {
1973 // The chain encoded the full key. Reaching a packed value
1974 // means the key already exists (duplicate). The ART prefix
1975 // restriction guarantees no key is a prefix of another.
1976 UNODB_DETAIL_ASSERT(remaining_key.size() == 1);
1977 return false;
1978 }
1979 }
1980
1981 ++depth;
1982 remaining_key.shift_right(1);
1983 }

2372template <typename Key, typename Value, typename HeapTag>
2373typename db<Key, Value, HeapTag>::iterator&
2374db<Key, Value, HeapTag>::iterator::seek(art_key_type search_key, bool& match,
2375 bool fwd) {
2376 invalidate(); // invalidate the iterator (clear the stack).
2377 match = false; // unless we wind up with an exact match.
2378 if (UNODB_DETAIL_UNLIKELY(db_.root == nullptr)) return *this; // aka end
2379
2380 // Empty key_view sorts before all keys — go to first (fwd) or end (rev).
2381 if constexpr (std::is_same_v<Key, key_view>) {
2382 if (UNODB_DETAIL_UNLIKELY(search_key.size() == 0)) {
2383 return fwd ? left_most_traversal(db_.root) : *this;
2384 }
2385 }
2386
2387 auto node{db_.root};
2388 const auto k = search_key;
2389 auto remaining_key{k};
2390
2391 while (true) {
2392 const auto node_type = node.type();
2393 if constexpr (!art_policy::can_eliminate_leaf) {
2394 if (node_type == node_type::LEAF) {
2395 const auto* const leaf{node.template ptr<leaf_type*>()};
2396 push_leaf(node);
2397 int cmp_{0};
2398 if constexpr (art_policy::full_key_in_inode_path) {
2399 cmp_ =
2400 unodb::detail::compare(keybuf_.get_key_view(), k.get_key_view());
2401 } else {
2402 cmp_ = leaf->cmp(k);
2403 }
2404 if (cmp_ == 0) {
2405 match = true;
2406 return *this;
2407 }
2408 if (fwd) { // GTE semantics
2409 // if search_key < leaf, use the leaf, else next().
2410 return (cmp_ < 0) ? *this : next();
2411 }
2412 // LTE semantics: if search_key > leaf, use the leaf, else prior().
2413 return (cmp_ > 0) ? *this : prior();
2414 }
2415 } // if constexpr (!can_eliminate_leaf)
2416 UNODB_DETAIL_ASSERT(node_type != node_type::LEAF);
2417 auto* const inode{node.template ptr<inode_type*>()}; // some internal node.
2418 const auto key_prefix{inode->get_key_prefix().get_snapshot()}; // prefix
2419 const auto key_prefix_length{key_prefix.length()}; // length of that prefix
2420 const auto shared_length = key_prefix.get_shared_length(
2421 remaining_key.get_u64()); // #of prefix bytes matched.
2422 if (shared_length < key_prefix_length) {
2423 // We have visited an internal node whose prefix is longer than
2424 // the bytes in the key that we need to match. To figure out
2425 // whether the search key would be located before or after the
2426 // current internal node, we need to compare the respective key
2427 // spans lexicographically. Since we have [shared_length] bytes
2428 // in common, we know that the next byte will tell us the
2429 // relative ordering of the key vs the prefix. So now we compare
2430 // prefix and key and the first byte where they differ.
2431 const auto cmp_ = static_cast<int>(remaining_key[shared_length]) -
2432 static_cast<int>(key_prefix[shared_length]);
2433 UNODB_DETAIL_ASSERT(cmp_ != 0);
2434 if (fwd) {
2435 if (cmp_ < 0) {
2436 // FWD and the search key is ordered before this node. We
2437 // want the left-most leaf under the node.
2438 return left_most_traversal(node);
2439 }
2440 // FWD and the search key is ordered after this node. Right
2441 // most descent and then next().
2442 return right_most_traversal(node).next();
2443 }
2444 // reverse traversal
2445 if (cmp_ < 0) {
2446 // REV and the search key is ordered before this node. We
2447 // want the preceding key.
2448 return left_most_traversal(node).prior();
2449 }
2450 // REV and the search key is ordered after this node.
2451 return right_most_traversal(node);
2452 }
2453 remaining_key.shift_right(key_prefix_length);
2454 // Key fully consumed by prefix — search key is a proper prefix of
2455 // all keys under this inode. Leftmost leaf is GTE, prior is LTE.
2456 if constexpr (std::is_same_v<Key, key_view>) {
2457 if (UNODB_DETAIL_UNLIKELY(remaining_key.size() == 0)) {
2458 return fwd ? left_most_traversal(node)
2459 : left_most_traversal(node).prior();
2460 }
2461 }
2462 const auto res = inode->find_child(node_type, remaining_key[0]);
2463 if (res.second == nullptr) {
2464 // We are on a key byte during the descent that is not mapped by
2465 // the current node. Where we go next depends on whether we are
2466 // doing forward or reverse traversal.
2467 if (fwd) {
2468 // FWD: Take the next child_index that is mapped in the data
2469 // and then do a left-most descent to land on the key that is
2470 // the immediate successor of the desired key in the data.
2471 //
2472 // Note: We are probing with a key byte which does not appear
2473 // in our list of keys (this was verified above) so this will
2474 // always be the index the first entry whose key byte is
2475 // greater-than the probe value and [false] if there is no
2476 // such entry.
2477 //
2478 // Note: [node] has not been pushed onto the stack yet!
2479 auto nxt = inode->gte_key_byte(node_type, remaining_key[0]);
2480 if (!nxt) {
2481 // Pop entries off the stack until we find one with a
2482 // right-sibling of the path we took to this node and then
2483 // do a left-most descent under that right-sibling. If there
2484 // is no such parent, we will wind up with an empty stack
2485 // (iterator at the end) and return that state.
2486 //
2487 // Note: The current node (where gte_key_byte failed) was
2488 // never pushed, so the stack top is already the first
2489 // ancestor to check for a right-sibling.
2490 while (!empty()) {
2491 const auto& centry = top();
2492 const auto cnode{centry.node}; // possible parent from the stack
2493 auto* const icnode{cnode.template ptr<inode_type*>()};
2494 const auto cnxt = icnode->next(
2495 cnode.type(), centry.child_index); // right-sibling.
2496 if (cnxt) {
2497 const auto& e2 = cnxt.value();
2498 pop();
2499 push(e2);
2500 auto nchild = icnode->get_child(cnode.type(), e2.child_index);
2501 return descend_left(icnode, cnode.type(), e2.child_index, nchild);
2502 }
2503 pop();
2504 }
2505 return *this; // stack is empty (aka end iterator).
2506 }
2507 const auto& tmp = nxt.value(); // unwrap.
2508 const auto child_index = tmp.child_index;
2509 const auto child = inode->get_child(node_type, child_index);
2510 push(node, tmp.key_byte, child_index, tmp.prefix); // the path we took
2511 return descend_left(inode, node_type, child_index, child);
2512 }
2513 // REV: Take the prior child_index that is mapped and then do
2514 // a right-most descent to land on the key that is the
2515 // immediate predecessor of the desired key in the data.
2516 auto nxt = inode->lte_key_byte(node_type, remaining_key[0]);
2517 if (!nxt) {
2518 // Pop off entries until we find one with a left-sibling and
2519 // then do a right-most descent under that left-sibling. In
2520 // the extreme case there is no such previous entry and we
2521 // will wind up with an empty stack.
2522 //
2523 // Note: The current node (where lte_key_byte failed) was
2524 // never pushed, so the stack top is already the first
2525 // ancestor to check for a left-sibling.
2526 while (!empty()) {
2527 const auto& centry = top();
2528 const auto cnode{centry.node}; // possible parent from stack
2529 auto* const icnode{cnode.template ptr<inode_type*>()};
2530 const auto cnxt =
2531 icnode->prior(cnode.type(), centry.child_index); // left-sibling.
2532 if (cnxt) {
2533 const auto& e2 = cnxt.value();
2534 pop();
2535 push(e2);
2536 auto nchild = icnode->get_child(cnode.type(), e2.child_index);
2537 return descend_right(icnode, cnode.type(), e2.child_index, nchild);
2538 }
2539 pop();
2540 }
2541 return *this; // stack is empty (aka end iterator).
2542 }
2543 const auto& tmp = nxt.value(); // unwrap.
2544 const auto child_index{tmp.child_index};
2545 const auto child = inode->get_child(node_type, child_index);
2546 push(node, tmp.key_byte, child_index, tmp.prefix); // the path we took
2547 return descend_right(inode, node_type, child_index, child);
2548 }
2549 // Simple case. There is a child for the current key byte.
2550 const auto child_index{res.first};
2551 const auto* const child{res.second};
2552 push(node, remaining_key[0], child_index, key_prefix);
2553 if constexpr (art_policy::can_eliminate_leaf) {
2554 if (inode->is_value_in_slot(node_type, child_index)) {
2555 push_leaf(*child);
2556 if (remaining_key.size() <= 1) {
2557 match = true;
2558 return *this;
2559 }
2560 // VIS key is a strict prefix of search key (VIS < search).
2561 return fwd ? next() : *this;
2562 }
2563 }
2564 node = *child;
2565 remaining_key.shift_right(1);
2566 } // while ( true )
2567 UNODB_DETAIL_CANNOT_HAPPEN();
2568}
