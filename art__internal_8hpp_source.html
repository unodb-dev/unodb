51[[nodiscard, gnu::pure]] inline int compare(const void* a, const size_t alen,
52 const void* b,
53 const size_t blen) noexcept {
54 // TODO(thompsonbry) consider changing this over to std::span
55 // arguments and do not let the (ptr,len) pattern propagate
56 // outwards.
57 const auto shared_length = std::min(alen, blen);
58 auto ret = std::memcmp(a, b, shared_length);
59 ret = (ret != 0) ? ret : ((alen == blen) ? 0 : (alen < blen ? -1 : 1));
60 return ret;
61}

515 [[nodiscard, gnu::const]] static std::uintptr_t tag_ptr(
516 const Header* ptr_, unodb::node_type tag) noexcept {
517 const auto uintptr = reinterpret_cast<std::uintptr_t>(ptr_);
518 const auto result =

529 [[nodiscard, gnu::const]] static constexpr unsigned mask_bits_needed(
530 unsigned count) noexcept {
531 return count < 2 ? 1 : 1 + mask_bits_needed(count >> 1U);
532 }
533
535 static constexpr auto lowest_non_tag_bit =
536 1ULL << mask_bits_needed(node_type_count);
538 static constexpr auto tag_bit_mask = lowest_non_tag_bit - 1;
540 static constexpr auto ptr_bit_mask = ~tag_bit_mask;
541
542 // Node type tags occupy the low bits of pointers to nodes. Nodes are
543 // heap-allocated via alignment_for_new<T>() (>=
544 // __STDCPP_DEFAULT_NEW_ALIGNMENT__), so their alignment leaves the tag bits
545 // free even when the header base is under-aligned (e.g. the empty
546 // node_header, or the 4-aligned non-OLC leaf). The wrapper's own size is
547 // checked at the node_ptr / olc_node_ptr alias definitions, where a concrete
548 // Header is in scope.
549 static_assert(
550 alignment_for_new<Header>() >= lowest_non_tag_bit,
551 "Node allocation alignment must leave the node type tag bits free");
