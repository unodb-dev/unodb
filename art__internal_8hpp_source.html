48[[nodiscard, gnu::pure]] inline int compare(const void* a, const size_t alen,
49 const void* b,
50 const size_t blen) noexcept {
51 // TODO(thompsonbry) consider changing this over to std::span
52 // arguments and do not let the (ptr,len) pattern propagate
53 // outwards.
54 const auto shared_length = std::min(alen, blen);
55 auto ret = std::memcmp(a, b, shared_length);
56 ret = (ret != 0) ? ret : ((alen == blen) ? 0 : (alen < blen ? -1 : 1));
57 return ret;
58}

66 const unodb::key_view b) noexcept {

499 [[nodiscard, gnu::const]] static std::uintptr_t tag_ptr(
500 const Header* ptr_, unodb::node_type tag) noexcept {
501 const auto uintptr = reinterpret_cast<std::uintptr_t>(ptr_);
502 const auto result =
503 uintptr | static_cast<std::underlying_type_t<decltype(tag)>>(tag);
504 UNODB_DETAIL_ASSERT((result & ptr_bit_mask) == uintptr);
505 return result;

513 [[nodiscard, gnu::const]] static constexpr unsigned mask_bits_needed(
514 unsigned count) noexcept {
515 return count < 2 ? 1 : 1 + mask_bits_needed(count >> 1U);
516 }
517
519 static constexpr auto lowest_non_tag_bit =
520 1ULL << mask_bits_needed(node_type_count);
522 static constexpr auto tag_bit_mask = lowest_non_tag_bit - 1;
524 static constexpr auto ptr_bit_mask = ~tag_bit_mask;
525
526 // Node type tags occupy the low bits of pointers to nodes. Nodes are
527 // heap-allocated via alignment_for_new<T>() (>=
528 // __STDCPP_DEFAULT_NEW_ALIGNMENT__), so their alignment leaves the tag bits
529 // free even when the header base is under-aligned (e.g. the empty
530 // node_header, or the 4-aligned non-OLC leaf). The wrapper's own size is
531 // checked at the node_ptr / olc_node_ptr alias definitions, where a concrete
532 // Header is in scope.
533 static_assert(
534 alignment_for_new<Header>() >= lowest_non_tag_bit,
535 "Node allocation alignment must leave the node type tag bits free");
