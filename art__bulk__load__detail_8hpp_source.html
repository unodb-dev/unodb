51 [[nodiscard]] static std::size_t common_prefix_length(RandomIt f, RandomIt l,
52 tree_depth_type depth) {
53 const art_key_type first_ak{f->first};
54 const art_key_type last_ak{std::prev(l)->first};
55 const auto fk = first_ak.get_key_view();
56 const auto lk = last_ak.get_key_view();
57 const auto d = static_cast<std::size_t>(depth);
58 const auto max_len = std::min(fk.size(), lk.size());
59 std::size_t len = 0;
60 while (d + len < max_len && fk[d + len] == lk[d + len]) ++len;
61 return len;
62 }

65 [[nodiscard]] static boost::container::small_vector<partition_entry<RandomIt>,
66 16>
67 partition_by_byte(RandomIt f, RandomIt l, tree_depth_type dispatch_depth) {
68 boost::container::small_vector<partition_entry<RandomIt>, 16> parts;
69 const auto dd = static_cast<std::size_t>(dispatch_depth);
70 auto cur = f;
71 while (cur != l) {
72 const art_key_type ak{cur->first};
73 const auto kv = ak.get_key_view();
74 UNODB_DETAIL_ASSERT(dd < kv.size());
75 const auto byte = kv[dd];
76 parts.push_back({cur, byte});
77 ++cur;
78 while (cur != l) {
79 const art_key_type ak2{cur->first};
80 const auto kv2 = ak2.get_key_view();
81 UNODB_DETAIL_ASSERT(dd < kv2.size());
82 if (kv2[dd] != byte) break;
83 ++cur;
84 }
85 }
86 return parts;
87 }

91 [[nodiscard]] static node_ptr_t build_prefix_chain(
92 Db& self, art_key_type k, node_ptr_t child_inode,
93 tree_depth_type start_depth, std::size_t end_depth) {
94 constexpr std::size_t prefix_cap = key_prefix_capacity;
95 const auto full_key = k.get_key_view();
96 const auto start = static_cast<std::size_t>(start_depth);
97 auto current = child_inode;
98 std::size_t pos = end_depth;
99 while (pos > start + prefix_cap) {
100 const auto depth = pos - prefix_cap - 1;
101 const auto dispatch = full_key[pos - 1];
102 auto remaining = k;
103 remaining.shift_right(depth);
104 auto chain{
105 inode_4::create(self, full_key, remaining,
106 tree_depth_type{static_cast<std::uint32_t>(depth)},
107 dispatch, current)};
108 current = node_ptr_t{chain.release(), node_type::I4};
109#ifdef UNODB_DETAIL_WITH_STATS
110 self.template account_growing_inode<node_type::I4>();
111#endif
112 pos = depth;
113 }
114 // LCOV_EXCL_START — always-false: chain_consumed is always a multiple of
115 // (prefix_cap+1), so the while loop always decrements pos to exactly start.
116 UNODB_DETAIL_ASSERT(pos == start);
117 if (pos > start) {
118 const auto dispatch = full_key[pos - 1];
119 auto chain{inode_4::create(
120 self, full_key, tree_depth_type{static_cast<std::uint32_t>(start)},
121 static_cast<key_prefix_size>(pos - start - 1), dispatch, current)};
122 current = node_ptr_t{chain.release(), node_type::I4};
123#ifdef UNODB_DETAIL_WITH_STATS
124 self.template account_growing_inode<node_type::I4>();
125#endif
126 }
127 // LCOV_EXCL_STOP
128 return current;
129 }

132 [[nodiscard]] static build_result_t build_single_leaf(Db& self, RandomIt it,
133 tree_depth_type depth) {
134 const art_key_type ak{it->first};
135 if constexpr (art_policy::can_eliminate_leaf) {
136 const auto packed = art_policy::pack_value(it->second);
137 if (static_cast<std::size_t>(depth) >= ak.size()) {
138 return {packed, true};
139 }
140 return {self.build_chain(ak, packed, depth), false};
141 } else if constexpr (art_policy::can_eliminate_key_in_leaf) {
142 auto leaf = art_policy::make_db_leaf_ptr(ak, it->second, self);
143 auto leaf_ptr = node_ptr_t{leaf.release(), node_type::LEAF};
144 if (static_cast<std::size_t>(depth) >= ak.size()) {
145 return {leaf_ptr, false};
146 }
147 return {self.build_chain(ak, leaf_ptr, depth), false};
148 } else {
149 static_cast<void>(depth);
150 auto leaf = art_policy::make_db_leaf_ptr(ak, it->second, self);
151 const auto leaf_ptr = node_ptr_t{leaf.release(), node_type::LEAF};
152 return {leaf_ptr, false};
153 }
154 }

157 [[nodiscard]] static node_ptr_t make_inode(
158 Db& self, std::span<const bulk_child_t> cs,
159 const boost::container::small_vector<guard_t, 16>& guards,
160 const boost::container::small_vector<bulk_child_t, 16>& children,
161 key_prefix_size inode_prefix_len, key_view prefix_kv,
162 tree_depth_type inode_depth) {
163 if constexpr (!art_policy::can_eliminate_leaf) {
164 static_cast<void>(guards);
165 static_cast<void>(children);
166 }
167 const auto child_count = cs.size();
168 if (child_count <= 4) {
169 std::uint8_t vmask = 0;
170 if constexpr (art_policy::can_eliminate_leaf) {
171 for (std::size_t i = 0; i < child_count; ++i) {
172 if (guards[i].is_packed_value)
173 vmask |= static_cast<std::uint8_t>(1U << i);
174 }
175 }
176 auto ptr = inode_4::create_bulk(self, inode_prefix_len, prefix_kv,
177 inode_depth, cs, vmask);
178#ifdef UNODB_DETAIL_WITH_STATS
179 self.template account_growing_inode<node_type::I4>();
180#endif
181 return node_ptr_t{ptr.release(), node_type::I4};
182 }
183 if (child_count <= 16) {
184 std::uint16_t vmask = 0;
185 if constexpr (art_policy::can_eliminate_leaf) {
186 for (std::size_t i = 0; i < child_count; ++i) {
187 if (guards[i].is_packed_value)
188 vmask |= static_cast<std::uint16_t>(1U << i);
189 }
190 }
191 auto ptr = inode_16::create_bulk(self, inode_prefix_len, prefix_kv,
192 inode_depth, cs, vmask);
193#ifdef UNODB_DETAIL_WITH_STATS
194 self.template account_growing_inode<node_type::I16>();
195#endif
196 return node_ptr_t{ptr.release(), node_type::I16};
197 }
198 if (child_count <= 48) {
199 std::array<std::uint8_t, 6> vmask{};
200 if constexpr (art_policy::can_eliminate_leaf) {
201 for (std::size_t i = 0; i < child_count; ++i) {
202 if (guards[i].is_packed_value)
203 vmask[i / 8] |= static_cast<std::uint8_t>(1U << (i % 8));
204 }
205 }
206 auto ptr = inode_48::create_bulk(self, inode_prefix_len, prefix_kv,
207 inode_depth, cs, vmask);
208#ifdef UNODB_DETAIL_WITH_STATS
209 self.template account_growing_inode<node_type::I48>();
210#endif
211 return node_ptr_t{ptr.release(), node_type::I48};
212 }
213 std::array<std::uint8_t, 32> vmask{};
214 if constexpr (art_policy::can_eliminate_leaf) {
215 for (std::size_t i = 0; i < child_count; ++i) {
216 if (guards[i].is_packed_value) {
217 const auto kb = static_cast<std::uint8_t>(children[i].key_byte);
218 vmask[static_cast<std::size_t>(kb / 8)] |=
219 static_cast<std::uint8_t>(1U << (kb % 8));
220 }
221 }
222 }
223 auto ptr = inode_256::create_bulk(self, inode_prefix_len, prefix_kv,
224 inode_depth, cs, vmask);
225#ifdef UNODB_DETAIL_WITH_STATS
226 self.template account_growing_inode<node_type::I256>();
227#endif
228 return node_ptr_t{ptr.release(), node_type::I256};
229 }

232 [[nodiscard]] static build_result_t build_subtree(Db& self, RandomIt f,
233 RandomIt l,
234 tree_depth_type depth) {
235 constexpr std::size_t prefix_cap = key_prefix_capacity;
236
237 const auto n = std::distance(f, l);
238 if (n == 0) return {node_ptr_t{nullptr}, false};
239 if (n == 1) return build_single_leaf(self, f, depth);
240
241 const auto prefix_len = common_prefix_length(f, l, depth);
242 const auto dispatch_depth = tree_depth_type{static_cast<std::uint32_t>(
243 static_cast<std::size_t>(depth) + prefix_len)};
244 auto parts = partition_by_byte(f, l, dispatch_depth);
245 const auto child_count = parts.size();
246
247 boost::container::small_vector<guard_t, 16> guards;
248 guards.reserve(child_count);
249 boost::container::small_vector<bulk_child_t, 16> children;
250 children.reserve(child_count);
251
252 const auto next_depth = tree_depth_type{static_cast<std::uint32_t>(
253 static_cast<std::size_t>(dispatch_depth) + 1)};
254
255 for (std::size_t i = 0; i < child_count; ++i) {
256 const auto part_begin = parts[i].begin;
257 const auto part_end = (i + 1 < child_count) ? parts[i + 1].begin : l;
258 auto result = build_subtree(self, part_begin, part_end, next_depth);
259 guards.emplace_back(self);
260 guards.back().ptr = result.ptr;
261 guards.back().is_packed_value = result.is_packed_value;
262 children.push_back({parts[i].key_byte, result.ptr});
263 }
264
265 const std::size_t chain_consumed =
266 (prefix_len > prefix_cap)
267 ? (prefix_len / (prefix_cap + 1)) * (prefix_cap + 1)
268 : 0;
269 const auto inode_prefix_len =
270 static_cast<key_prefix_size>(prefix_len - chain_consumed);
271 const auto inode_depth = tree_depth_type{static_cast<std::uint32_t>(
272 static_cast<std::size_t>(depth) + chain_consumed)};
273
274 const art_key_type prefix_key{f->first};
275 const auto prefix_kv = prefix_key.get_key_view();
276
277 const auto cs =
278 std::span<const bulk_child_t>{children.data(), children.size()};
279 auto inode_ptr = make_inode(self, cs, guards, children, inode_prefix_len,
280 prefix_kv, inode_depth);
281 for (auto& g : guards) g.release();
282
283 if (chain_consumed > 0) {
284 return {
285 build_prefix_chain(self, prefix_key, inode_ptr, depth,
286 chain_consumed + static_cast<std::size_t>(depth)),
287 false};
288 }
289 return {inode_ptr, false};
290 }

301template <typename Db, typename Fork, typename RandomIt>
302void bulk_load_impl(Db& self, Fork&& fork, std::size_t max_tasks,
303 RandomIt first, RandomIt last) {
304 using H = bulk_load_helpers<Db, RandomIt>;
305 using build_result_t = typename Db::build_result;
306 using guard_t = typename Db::bulk_subtree_guard;
307 using node_ptr_t = typename build_result_t::ptr_type;
308 using bulk_child_t = bulk_child<node_ptr_t>;
309 using tree_depth_type = typename Db::tree_depth_type;
310 using art_key_type = typename Db::art_key_type;
311 constexpr std::size_t prefix_cap = key_prefix_capacity;
312
313 constexpr bool is_parallel =
314 !std::is_same_v<std::remove_cvref_t<Fork>, std::nullptr_t>;
315
316 if constexpr (!is_parallel) {
317 auto result = H::build_subtree(self, first, last, tree_depth_type{0});
318 self.root = result.ptr;
319 } else {
320 const auto n = std::distance(first, last);
321 if (n <= 1) {
322 auto result = H::build_subtree(self, first, last, tree_depth_type{0});
323 self.root = result.ptr;
324 return;
325 }
326
327 const auto prefix_len =
328 H::common_prefix_length(first, last, tree_depth_type{0});
329 const auto dispatch_depth =
330 tree_depth_type{static_cast<std::uint32_t>(prefix_len)};
331 auto parts = H::partition_by_byte(first, last, dispatch_depth);
332 const auto child_count = parts.size();
333
334 const auto next_depth = tree_depth_type{static_cast<std::uint32_t>(
335 static_cast<std::size_t>(dispatch_depth) + 1)};
336
337 // Fork up to min(child_count, max_tasks) subtrees; inline the rest.
338 const auto fork_count = std::min(child_count, max_tasks);
339
340#ifdef UNODB_DETAIL_WITH_STATS
341 struct parallel_result {
342 build_result_t build;
343 bulk_load_stats_accumulator stats;
344 };
345#else
346 using parallel_result = build_result_t;
347#endif
348
349 auto make_task = [&self, &parts, child_count, last,
350 next_depth](std::size_t i) {
351 const auto part_begin = parts[i].begin;
352 const auto part_end = (i + 1 < child_count) ? parts[i + 1].begin : last;
353#ifdef UNODB_DETAIL_WITH_STATS
354 bulk_load_stats_accumulator acc;
355 tls_bulk_load_stats = &acc;
356 auto result = H::build_subtree(self, part_begin, part_end, next_depth);
357 tls_bulk_load_stats = nullptr;
358 return parallel_result{result, acc};
359#else
360 return H::build_subtree(self, part_begin, part_end, next_depth);
361#endif
362 };
363
364 // Submit forked tasks — first fork deduces the future type.
365 auto first_future =
366 fork([&make_task] { return make_task(std::size_t{0}); });
367 using future_t = decltype(first_future);
368 std::vector<future_t> futures;
369 futures.reserve(fork_count);
370 futures.push_back(std::move(first_future));
371 for (std::size_t i = 1; i < fork_count; ++i) {
372 futures.push_back(fork([&make_task, i] { return make_task(i); }));
373 }
374
375 // Run remaining partitions inline on the calling thread.
376 std::vector<parallel_result> inline_results;
377 inline_results.reserve(child_count - fork_count);
378 for (std::size_t i = fork_count; i < child_count; ++i) {
379 inline_results.push_back(make_task(i));
380 }
381
382 boost::container::small_vector<guard_t, 16> guards;
383 guards.reserve(child_count);
384 boost::container::small_vector<bulk_child_t, 16> children;
385 children.reserve(child_count);
386
387 // Gather forked results first (indices 0..fork_count-1).
388 for (std::size_t i = 0; i < fork_count; ++i) {
389 try {
390 auto future_result = futures[i].get();
391#ifdef UNODB_DETAIL_WITH_STATS
392 const auto& result = future_result.build;
393 const auto& acc = future_result.stats;
394 self.merge_bulk_load_stats(acc);
395#else
396 const auto& result = future_result;
397#endif
398 guards.emplace_back(self);
399 guards.back().ptr = result.ptr;
400 guards.back().is_packed_value = result.is_packed_value;
401 children.push_back({parts[i].key_byte, result.ptr});
402 } catch (...) { // LCOV_EXCL_START — exception drain; no reliable test
403 // Drain remaining futures into guards to prevent leaks.
404 for (std::size_t j = i + 1; j < fork_count; ++j) {
405 try {
406 auto r = futures[j].get();
407 guards.emplace_back(self);
408#ifdef UNODB_DETAIL_WITH_STATS
409 guards.back().ptr = r.build.ptr;
410 guards.back().is_packed_value = r.build.is_packed_value;
411#else
412 guards.back().ptr = r.ptr;
413 guards.back().is_packed_value = r.is_packed_value;
414#endif
415 } catch (...) { // future.get() may rethrow task exception
416 }
417 }
418 throw;
419 } // LCOV_EXCL_STOP
420 }
421
422 // Gather inline results (indices fork_count..child_count-1).
423 for (std::size_t i = 0; i < inline_results.size(); ++i) {
424 const auto part_idx = fork_count + i;
425#ifdef UNODB_DETAIL_WITH_STATS
426 const auto& result = inline_results[i].build;
427 const auto& acc = inline_results[i].stats;
428 self.merge_bulk_load_stats(acc);
429#else
430 const auto& result = inline_results[i];
431#endif
432 guards.emplace_back(self);
433 guards.back().ptr = result.ptr;
434 guards.back().is_packed_value = result.is_packed_value;
435 children.push_back({parts[part_idx].key_byte, result.ptr});
436 }
437
438 const std::size_t chain_consumed =
439 (prefix_len > prefix_cap)
440 ? (prefix_len / (prefix_cap + 1)) * (prefix_cap + 1)
441 : 0;
442 const auto inode_prefix_len =
443 static_cast<key_prefix_size>(prefix_len - chain_consumed);
444 const auto inode_depth =
445 tree_depth_type{static_cast<std::uint32_t>(chain_consumed)};
446
447 const art_key_type prefix_key{first->first};
448 const auto prefix_kv = prefix_key.get_key_view();
449
450 const auto cs =
451 std::span<const bulk_child_t>{children.data(), children.size()};
452 auto inode_ptr = H::make_inode(self, cs, guards, children, inode_prefix_len,
453 prefix_kv, inode_depth);
454 for (auto& g : guards) g.release();
455
456 if (chain_consumed > 0) {
457 self.root = H::build_prefix_chain(self, prefix_key, inode_ptr,
458 tree_depth_type{0}, chain_consumed);
459 } else {
460 self.root = inode_ptr;
461 }
462 }
463}
