44 [[nodiscard]] get_result get_internal(art_key_type k) const noexcept {
45 std::unique_lock guard{mutex};
46 const auto db_get_result{db_.get_internal(k)};
47 if (!db_get_result) {
48 guard.unlock();
49 return std::make_pair(db_get_result, std::unique_lock<std::mutex>{});
50 }
51 return std::make_pair(db_get_result, std::move(guard));
52 }

307 // Releases the mutex in the case key was not found, keeps it locked
308 // otherwise.
309 [[nodiscard]] static bool key_found(const get_result& result) noexcept {
310#ifndef NDEBUG
311 const auto& lock{result.second};
312 // NOLINTNEXTLINE(readability-simplify-boolean-expr)
313 assert(!result.first || lock.owns_lock());
314#endif
315
316 return static_cast<bool>(result.first);
317 }
