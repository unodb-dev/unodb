66 static void maybe_fail() {
67 // Inspects the fail counter. If non-zero, then bumps the allocation
68 // counter. If that results in the allocation counter reaching or exceeding
69 // the fail counter, then throw std::bad_alloc.
70 if (UNODB_DETAIL_UNLIKELY(paused)) return;
71 const auto fail_counter =
72 fail_on_nth_allocation_.load(std::memory_order_acquire);
73 if (UNODB_DETAIL_UNLIKELY(fail_counter != 0) &&
74 (allocation_counter.fetch_add(1, std::memory_order_relaxed) >=
75 fail_counter - 1)) {
76 throw std::bad_alloc{};
77 }
78 }
