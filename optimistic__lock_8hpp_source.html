390 void write_unlock() noexcept {
391 // This thread has written the previous lock word value, and no other
392 // thread may write it before the unlock, thus we can read it without
393 // ordering.
394 const auto old_lock_word = load_relaxed();
395 UNODB_DETAIL_ASSERT(old_lock_word.is_write_locked());
396
397 const auto new_lock_word = old_lock_word.get() + 2;
398 version.store(new_lock_word, std::memory_order_release);
399 }

410#endif
411
412 version.store(version_type::obsolete_lock_word,
413 std::memory_order_release);

700 [[nodiscard]] read_critical_section try_read_lock() noexcept {
701 while (true) {
702 const auto current_version = version.load_acquire();
703 if (UNODB_DETAIL_LIKELY(current_version.is_free())) {
704 inc_read_lock_count();
705 return read_critical_section{*this, current_version};
706 }
707 // LCOV_EXCL_START
708 if (UNODB_DETAIL_UNLIKELY(current_version.is_obsolete()))
709 return read_critical_section{};
710 UNODB_DETAIL_ASSERT(current_version.is_write_locked());
711 spin_wait_loop_body();
712 // LCOV_EXCL_STOP
713 }
714 }

725 [[nodiscard]] read_critical_section try_read_lock_no_wait() noexcept {
726 const auto current_version = version.load_acquire();
727 if (UNODB_DETAIL_UNLIKELY(!current_version.is_free()))
728 return read_critical_section{};
729 inc_read_lock_count();
730 return read_critical_section{*this, current_version};
731 }
